    62→function cfTier(rank: string | null): { label: string; text: string } {
    63→  const r = (rank ?? "").toLowerCase();
    64→  if (r.includes("legendary"))
    65→    return { label: "LEGENDARY GM", text: "text-[oklch(0.72_0.25_320)]" };
    66→  if (r.includes("grandmaster"))
    67→    return { label: "GRANDMASTER", text: "text-[oklch(0.72_0.25_320/0.9)]" };
    68→  if (r.includes("master"))
    69→    return { label: r.replace(/\b\w/g, (c) => c.toUpperCase()), text: "text-[oklch(0.75_0.22_90)]" };
    70→  if (r.includes("expert"))
    71→    return { label: "EXPERT", text: "text-[oklch(0.78_0.20_195)]" };
    72→  if (r.includes("specialist"))
    73→    return { label: "SPECIALIST", text: "text-[oklch(0.78_0.20_195/0.85)]" };

   402→interface HeatWeek {
   403→  cells: { date: string; count: number; future: boolean }[];
   404→  monthLabel: string | null;
   405→}
   406→
   407→function dateKey(d: Date): string {
   408→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
   409→    d.getDate()
   410→  ).padStart(2, "0")}`;
   411→}
   412→
   413→/** Build the trailing 52-week grid (Sun-first columns, last col = current week). */
   414→function buildHeatWeeks(days: DayCount[]): HeatWeek[] {
   415→  const map = new Map(days.map((d) => [d.date, d.count]));
   416→  const today = new Date();
   417→  today.setHours(12, 0, 0, 0);
   418→  const lastSunday = new Date(today);
   419→  lastSunday.setDate(today.getDate() - today.getDay());
   420→
   421→  const weeks: HeatWeek[] = [];
   422→  let prevMonth = -1;
   423→  for (let w = HEAT_WEEKS - 1; w >= 0; w--) {
   424→    const weekStart = new Date(lastSunday);
   425→    weekStart.setDate(lastSunday.getDate() - w * 7);
   426→    const cells: { date: string; count: number; future: boolean }[] = [];
   427→    for (let dow = 0; dow < 7; dow++) {
   428→      const dt = new Date(weekStart);
   429→      dt.setDate(weekStart.getDate() + dow);
   430→      const future = dt > today;
   431→      cells.push({
   432→        date: future ? "" : dateKey(dt),
   433→        count: future ? 0 : (map.get(dateKey(dt)) ?? 0),
   434→        future,
   435→      });
   436→    }
   437→    const m = weekStart.getMonth();
   438→    const monthLabel = m !== prevMonth ? weekStart.toLocaleDateString("en", { month: "short" }) : null;
   439→    if (m !== prevMonth) prevMonth = m;
   440→    weeks.push({ cells, monthLabel });
   441→  }
   442→  return weeks;
   443→}

   445→function heatLevel(count: number, thresholds: number[]): number {
   446→  if (count <= 0) return 0;
   447→  if (count >= thresholds[2]) return 4;
   448→  if (count >= thresholds[1]) return 3;
   449→  if (count >= thresholds[0]) return 2;
   450→  return 1;
   451→}

   453→function MiniHeatmap({
   454→  days,
   455→  thresholds,
   456→  unit,
   457→  palette,
   458→}: {
   459→  days: DayCount[];
   460→  thresholds: number[];
   461→  unit: string;
   462→  palette: HeatPalette;
   463→}) {
   464→  const weeks = useMemo(() => buildHeatWeeks(days), [days]);
   465→  const activeTotal = days.reduce((s, d) => s + d.count, 0);
   466→  const levels = HEAT_PALETTES[palette];
   467→
   468→  return (
   469→    <div>
   470→      <div className="w-fit">
   471→        {/* month labels */}
   472→        <div className="relative mb-1 h-[8px]" aria-hidden>
   473→          {weeks.map((wk, wi) =>
   474→            wk.monthLabel ? (
   475→              <span
   476→                key={wk.cells[0].date || wi}
   477→                className="absolute font-mono text-[7px] leading-none text-muted-foreground/60"
   478→                style={{ left: `${wi * HEAT_STEP}px` }}
   479→              >
   480→                {wk.monthLabel}
   481→              </span>
   482→            ) : null
   483→          )}
   484→        </div>
   485→        {/* grid */}

   533→function relTime(iso: string): string {
   534→  const t = new Date(iso).getTime();
   535→  if (Number.isNaN(t)) return "—";
   536→  const diff = Math.max(0, Date.now() - t);
   537→  const m = Math.floor(diff / 60000);
   538→  if (m < 1) return "now";
   539→  if (m < 60) return `${m}m`;
   540→  const h = Math.floor(m / 60);
   541→  if (h < 24) return `${h}h`;
   542→  const d = Math.floor(h / 24);
   543→  if (d < 30) return `${d}d`;
   544→  return `${Math.floor(d / 30)}mo`;
   545→}

   718→/** SVG rating-trajectory chart (client-rendered from fetched data). */
   719→function RatingTrajectory({ history }: { history: CodeforcesHistoryPoint[] }) {
   720→  if (history.length === 0) {
   721→    return (
   722→      <div className="flex h-[64px] items-center justify-center rounded-lg border border-dashed border-white/10 bg-white/[0.02] font-mono text-[8px] tracking-[0.18em] text-muted-foreground/60">
   723→        NO CONTESTS LOGGED — TRAJECTORY ARMED
   724→      </div>
   725→    );
   726→  }
   727→
   728→  const W = 320;
   729→  const H = 88;
   730→  const PX = 10;
   731→  const TOP = 12;
   732→  const BOT = 78;
   733→  const n = history.length;
   734→  const ratings = history.map((h) => h.rating);
   735→  let min = Math.min(...ratings);
   736→  let max = Math.max(...ratings);
   737→  if (min === max) {
   738→    min -= 50;
   739→    max += 50;
   740→  } else {
   741→    const pad = (max - min) * 0.08;
   742→    min -= pad;
   743→    max += pad;
   744→  }
   745→
   746→  const x = (i: number) => (n === 1 ? W / 2 : PX + (i / (n - 1)) * (W - 2 * PX));
   747→  const y = (r: number) => BOT - ((r - min) / (max - min)) * (BOT - TOP);
   748→
   749→  const r2 = (v: number) => v.toFixed(2);
   750→  const line = history
   751→    .map((h, i) => `${i === 0 ? "M" : "L"}${r2(x(i))},${r2(y(h.rating))}`)
   752→    .join(" ");
   753→  const area = `${line} L${r2(x(n - 1))},${BOT} L${r2(x(0))},${BOT} Z`;
   754→  const peakIdx = ratings.indexOf(Math.max(...ratings));
   755→  const lastIdx = n - 1;
   756→  // Label side depends on peak position so text never clips the viewBox.
   757→  const peakLeft = x(peakIdx) < W / 2;
   758→  const peakTextX = peakLeft ? x(peakIdx) + 6 : x(peakIdx) - 6;
   759→
   760→  return (
   761→    <svg
   762→      viewBox={`0 0 ${W} ${H}`}
   763→      className="h-auto w-full"
   764→      role="img"
   765→      aria-label={`Codeforces rating history across ${n} contests`}
   766→    >
   767→      <defs>
   768→        <linearGradient id="cfAreaGrad" x1="0" y1="0" x2="0" y2="1">
   769→          <stop offset="0%" stopColor="oklch(0.78 0.20 195)" stopOpacity="0.3" />
   770→          <stop offset="100%" stopColor="oklch(0.72 0.25 320)" stopOpacity="0.02" />
   771→        </linearGradient>
   772→      </defs>
   773→      {/* baseline */}
   774→      <line
   775→        x1={PX}
   776→        y1={BOT}
   777→        x2={W - PX}
   778→        y2={BOT}
   779→        stroke="oklch(1 0 0 / 0.08)"
   780→        strokeWidth="1"
   781→      />
   782→      <path d={area} fill="url(#cfAreaGrad)" />
   783→      <motion.path
   784→        d={line}
   785→        fill="none"
   786→        stroke="oklch(0.82 0.22 165)"
   787→        strokeWidth="1.5"
   788→        strokeLinejoin="round"
   789→        strokeLinecap="round"
   790→        initial={{ pathLength: 0 }}
   791→        animate={{ pathLength: 1 }}
   792→        transition={{ duration: 1.4, ease: [0.22, 1, 0.36, 1] }}
   793→      />
   794→      {/* data points (with tooltips when sparse) */}
   795→      {history.map((h, i) =>
   796→        n <= 24 || i === peakIdx || i === lastIdx ? (
   797→          <circle
   798→            key={i}
   799→            cx={r2(x(i))}
   800→            cy={r2(y(h.rating))}
   801→            r="1.5"
   802→            fill="oklch(0.82 0.22 165 / 0.75)"
   803→          >
   804→            <title>{`${h.contest} · ${h.rating} · ${h.date.slice(0, 10)}`}</title>
   805→          </circle>
   806→        ) : null
   807→      )}
   808→      {/* peak marker */}
   809→      <circle
   810→        cx={r2(x(peakIdx))}
   811→        cy={r2(y(ratings[peakIdx]))}
   812→        r="2.5"
   813→        fill="oklch(0.72 0.25 320)"
   814→      >
   815→        <title>{`Peak ${ratings[peakIdx]} · ${history[peakIdx].contest}`}</title>
   816→      </circle>
   817→      {/* latest marker */}
   818→      <circle
   819→        cx={r2(x(lastIdx))}
   820→        cy={r2(y(ratings[lastIdx]))}
   821→        r="2.5"
   822→        fill="oklch(0.82 0.22 165)"
   823→      >
   824→        <title>{`Latest ${ratings[lastIdx]} · ${history[lastIdx].contest}`}</title>
   825→      </circle>
   826→      <text
   827→        x={r2(peakTextX)}
   828→        y={TOP}
   829→        textAnchor={peakLeft ? "start" : "end"}
   830→        fontSize="7"
   831→        fontFamily="ui-monospace, monospace"
   832→        fill="oklch(0.72 0.25 320)"
   833→      >
   834→        PEAK {ratings[peakIdx]}
   835→      </text>
   836→    </svg>
   837→  );
   838→}
