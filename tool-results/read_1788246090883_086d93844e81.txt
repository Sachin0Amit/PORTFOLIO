    62→    62→function cfTier(rank: string | null): { label: string; text: string } {
    63→    63→  const r = (rank ?? "").toLowerCase();
    64→    64→  if (r.includes("legendary"))
    65→    65→    return { label: "LEGENDARY GM", text: "text-[oklch(0.72_0.25_320)]" };
    66→    66→  if (r.includes("grandmaster"))
    67→    67→    return { label: "GRANDMASTER", text: "text-[oklch(0.72_0.25_320/0.9)]" };
    68→    68→  if (r.includes("master"))
    69→    69→    return { label: r.replace(/\b\w/g, (c) => c.toUpperCase()), text: "text-[oklch(0.75_0.22_90)]" };
    70→    70→  if (r.includes("expert"))
    71→    71→    return { label: "EXPERT", text: "text-[oklch(0.78_0.20_195)]" };
    72→    72→  if (r.includes("specialist"))
    73→    73→    return { label: "SPECIALIST", text: "text-[oklch(0.78_0.20_195/0.85)]" };
    74→    74→  if (r.includes("pupil"))
    75→    75→    return { label: "PUPIL", text: "text-foreground/85" };
    76→    76→  return { label: "NEWCOMER", text: "text-muted-foreground" };
    77→    77→}

    85→    85→export function Platforms() {
    86→    86→  const [data, setData] = useState<PlatformResponse[] | null>(null);
    87→    87→  const [loading, setLoading] = useState(true);
    88→    88→  const [refreshing, setRefreshing] = useState(false);
    89→    89→  const [lastSync, setLastSync] = useState<Date | null>(null);
    90→    90→
    91→    91→  const load = useCallback(async (manual = false) => {
    92→    92→    if (manual) setRefreshing(true);
    93→    93→    try {
    94→    94→      const res = await fetch("/api/platforms", { cache: "no-store" });
    95→    95→      const json = (await res.json()) as {
    96→    96→        ok: boolean;
    97→    97→        platforms: PlatformResponse[];
    98→    98→      };
    99→    99→      if (json.ok) {
   100→   100→        setData(json.platforms);
   101→   101→        setLastSync(new Date());
   102→   102→      }
   103→   103→    } catch {
   104→   104→      /* keep previous data on transient failure */
   105→   105→    } finally {
   106→   106→      setLoading(false);
   107→   107→      if (manual) setRefreshing(false);

   109→   109→  }, []);

   111→   111→  useEffect(() => {
   112→   112→    load();
   113→   113→    const id = setInterval(() => load(), 10 * 60 * 1000);
   114→   114→    return () => clearInterval(id);
   115→   115→  }, [load]);

   117→   117→  const byPlatform = new Map((data ?? []).map((p) => [p.platform, p]));
   118→   118→  const degradedCount = (data ?? []).filter(
   119→   119→    (p) => p.status === "degraded" || p.status === "disabled"
   120→   120→  ).length;
   121→   121→  const connectedCount = (data ?? []).filter((p) => p.connected).length;
   122→   122→  const syncTime = lastSync
   123→   123→    ? lastSync.toLocaleTimeString("en-GB", { hour12: false })
   124→   124→    : "--:--:--";
   125→   125→
   126→   126→  return (
   127→   127→    <section id="platforms" className="relative px-4 py-20 sm:py-28">
   128→   128→      {/* Ambient glow */}

   130→   130→
   131→   131→      <div className="mx-auto max-w-5xl">
   132→   132→        <SectionHeading
   133→   133→          index="07 // UPLINK"
   134→   134→          title="CODING UPLINK"
   135→   135→          subtitle="Live signals from LeetCode, Codeforces and GitHub — synced straight from each platform's public API."
   136→   136→          accent="cyan"
   137→   137→        />
   138→   138→
   139→   139→        <div className="grid gap-5 md:grid-cols-2 xl:grid-cols-3">
   140→   140→          {ORDER.map((id, i) => (
   141→   141→            <PlatformCard
   142→   142→              key={id}
   143→   143→              index={i}
   144→   144→              data={byPlatform.get(id) ?? null}
   145→   145→              loading={loading}
   146→   146→            />
   147→   147→          ))}
   148→   148→        </div>
   149→   149→
   150→   150→        {/* Sync strip */}
   151→   151→        <motion.div
   152→   152→          initial={{ opacity: 0, y: 16 }}
   153→   153→          whileInView={{ opacity: 1, y: 0 }}
   154→   154→          viewport={{ once: true }}
   155→   155→          transition={{ duration: 0.5, delay: 0.2 }}
   156→   156→          className="mt-5 flex flex-wrap items-center justify-between gap-3 rounded-xl border border-[oklch(0.82_0.22_165/0.15)] bg-[oklch(0.09_0.02_280/85)] px-4 py-2.5 backdrop-blur-md"

   407→   407→function dateKey(d: Date): string {
   408→   408→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(
   409→   409→    d.getDate()
   410→   410→  ).padStart(2, "0")}`;
   411→   411→}

   413→   413→/** Build the trailing 52-week grid (Sun-first columns, last col = current week). */
   414→   414→function buildHeatWeeks(days: DayCount[]): HeatWeek[] {
   415→   415→  const map = new Map(days.map((d) => [d.date, d.count]));
   416→   416→  const today = new Date();
   417→   417→  today.setHours(12, 0, 0, 0);
   418→   418→  const lastSunday = new Date(today);
   419→   419→  lastSunday.setDate(today.getDate() - today.getDay());
   420→   420→
   421→   421→  const weeks: HeatWeek[] = [];
   422→   422→  let prevMonth = -1;
   423→   423→  for (let w = HEAT_WEEKS - 1; w >= 0; w--) {
   424→   424→    const weekStart = new Date(lastSunday);
   425→   425→    weekStart.setDate(lastSunday.getDate() - w * 7);
   426→   426→    const cells: { date: string; count: number; future: boolean }[] = [];
   427→   427→    for (let dow = 0; dow < 7; dow++) {
   428→   428→      const dt = new Date(weekStart);
   429→   429→      dt.setDate(weekStart.getDate() + dow);
   430→   430→      const future = dt > today;
   431→   431→      cells.push({
   432→   432→        date: future ? "" : dateKey(dt),
   433→   433→        count: future ? 0 : (map.get(dateKey(dt)) ?? 0),
   434→   434→        future,
   435→   435→      });
   436→   436→    }
   437→   437→    const m = weekStart.getMonth();
   438→   438→    const monthLabel = m !== prevMonth ? weekStart.toLocaleDateString("en", { month: "short" }) : null;
   439→   439→    if (m !== prevMonth) prevMonth = m;
   440→   440→    weeks.push({ cells, monthLabel });
   441→   441→  }
   442→   442→  return weeks;
   443→   443→}

   445→   445→function heatLevel(count: number, thresholds: number[]): number {
   446→   446→  if (count <= 0) return 0;
   447→   447→  if (count >= thresholds[2]) return 4;
   448→   448→  if (count >= thresholds[1]) return 3;
   449→   449→  if (count >= thresholds[0]) return 2;
   450→   450→  return 1;
   451→   451→}

   533→   533→function relTime(iso: string): string {
   534→   534→  const t = new Date(iso).getTime();

   536→   536→  const diff = Math.max(0, Date.now() - t);
   537→   537→  const m = Math.floor(diff / 60000);
   538→   538→  if (m < 1) return "now";
   539→   539→  if (m < 60) return `${m}m`;
   540→   540→  const h = Math.floor(m / 60);
   541→   541→  if (h < 24) return `${h}h`;
   542→   542→  const d = Math.floor(h / 24);
   543→   543→  if (d < 30) return `${d}d`;
   544→   544→  return `${Math.floor(d / 30)}mo`;
   545→   545→}

   738→   738→    min -= 50;
   739→   739→    max += 50;

   743→   743→    max += pad;
